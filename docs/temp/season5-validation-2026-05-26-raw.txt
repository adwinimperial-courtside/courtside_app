B. Score-vs-stat-sum integrity (only MISMATCH rows shown)

-- SQL --

WITH sums AS (
  SELECT ps.game_id, ps.team_id,
         SUM(ps.points_2 * 2 + ps.points_3 * 3 + ps.free_throws)::int AS stat_sum
  FROM player_stats ps
  WHERE ps.league_id = '30639acb-a877-4068-9181-a857675106f6'
  GROUP BY ps.game_id, ps.team_id
)
SELECT g.id AS game_id, g.scheduled_at::date AS game_date,
       g.home_team_id, g.home_score, h.stat_sum AS home_sum,
       g.away_team_id, g.away_score, a.stat_sum AS away_sum
FROM games g
LEFT JOIN sums h ON h.game_id = g.id AND h.team_id = g.home_team_id
LEFT JOIN sums a ON a.game_id = g.id AND a.team_id = g.away_team_id
WHERE g.league_id = '30639acb-a877-4068-9181-a857675106f6'
  AND (COALESCE(h.stat_sum, 0) <> g.home_score OR COALESCE(a.stat_sum, 0) <> g.away_score)
ORDER BY g.scheduled_at;
